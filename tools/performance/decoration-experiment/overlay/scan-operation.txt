        if (type == EnvironmentScanPlacement.class) {
            try {
                var lookup=MethodHandles.privateLookupIn(type,MethodHandles.lookup());
                var direction=lookup.findGetter(type,"directionOfSearch",net.minecraft.core.Direction.class);
                var target=lookup.findGetter(type,"targetCondition",net.minecraft.world.level.levelgen.blockpredicates.BlockPredicate.class);
                var allowed=lookup.findGetter(type,"allowedSearchCondition",net.minecraft.world.level.levelgen.blockpredicates.BlockPredicate.class);
                var steps=lookup.findGetter(type,"maxSteps",int.class);
                return (modifier,run,next,origin)-> {
                    if (!DecorationExperimentSettings.scan) { OPAQUE.apply(modifier,run,next,origin);return; }
                    DecorationExperimentSettings.scans++;
                    try {
                        var pos=origin.mutable();
                        var level=run.context.getLevel();
                        var d=(net.minecraft.core.Direction)direction.invoke(modifier);
                        var t=(net.minecraft.world.level.levelgen.blockpredicates.BlockPredicate)target.invoke(modifier);
                        var a=(net.minecraft.world.level.levelgen.blockpredicates.BlockPredicate)allowed.invoke(modifier);
                        int maximum=(int)steps.invoke(modifier);
                        if(!a.test(level,pos))return;
                        for(int i=0;i<maximum;i++) {
                            if(t.test(level,pos)) {run.apply(next,pos.immutable());return;}
                            pos.move(d);
                            if(level.isOutsideBuildHeight(pos.getY()))return;
                            if(!a.test(level,pos))break;
                        }
                        if(t.test(level,pos))run.apply(next,pos.immutable());
                    } catch(RuntimeException|Error e) {throw e;} catch(Throwable e) {throw new IllegalStateException(e);}
                };
            } catch (ReflectiveOperationException e) { throw new IllegalStateException("Experiment scan layout changed",e); }
        }
